   132→   132→/**
   133→   133→ * Product assurance items — the short trust statements rendered in the
   134→   134→ * compact band directly beneath each product hero.
   135→   135→ *
   136→   136→ * ### Why stable IDs?
   137→   137→ *
   138→   138→ * The previous implementation stored trust statements as plain strings and
   139→   139→ * merged two uncontrolled arrays in the hero component — `product.trustPoints`
   140→   140→ * plus a hard-coded `staticTrustIndicators` list. Both arrays contained a
   141→   141→ * documentation entry (one said "Technical documentation available", the
   142→   142→ * other said "Technical documentation available on request"), and because
   143→   143→ * they were only compared as raw strings, both rendered at once on
   144→   144→ * BharatMembrane. That is how the screenshot ended up with two
   145→   145→ * "TECHNICAL DOCUMENTATION AVAILABLE" labels above each other.
   146→   146→ *
   147→   147→ * Stable semantic IDs fix this at the data layer. Every assurance item
   148→   148→ * belongs to exactly one category. When product-specific items are merged
   149→   149→ * with the shared defaults, deduplication happens by ID — so a product
   150→   150→ * cannot end up with two documentation entries even if its own data lists
   151→   151→ * one and the defaults also list one.
   152→   152→ *
   153→   153→ * ### Why per-product rather than global?
   154→   154→ *
   155→   155→ * The mat range is made under IS 15652:2006 and a BIS licence; the
   156→   156→ * geo-membrane is not. Applying one array to every product is how
   157→   157→ * `IS 15652:2006 Certified` previously ended up on BharatMembrane.
   158→   158→ */

   220→   220→/**
   221→   221→ * Merge a product's own assurance items with the shared defaults,
   222→   222→ * deduplicating by `id`.
   223→   223→ *
   224→   224→ * Product-specific items come first (in their declared order), then any
   225→   225→ * default item whose `id` is not already present is appended. This means a
   226→   226→ * product can override a default by re-declaring the same ID with a
   227→   227→ * different label, and no product can ever render two items with the same
   228→   228→ * category.
   229→   229→ */
   230→   230→export function mergeAssuranceItems(
   231→   231→  productItems: ProductAssurance[],
   232→   232→  defaults: ProductAssurance[] = defaultAssuranceItems
   233→   233→): ProductAssurance[] {
   234→   234→  const seen = new Set<AssuranceId>(productItems.map((item) => item.id));
   235→   235→  const merged: ProductAssurance[] = [...productItems];
   236→   236→  for (const def of defaults) {
   237→   237→    if (!seen.has(def.id)) {
   238→   238→      merged.push(def);
   239→   239→      seen.add(def.id);
   240→   240→    }
   241→   241→  }
   242→   242→  return merged;
   243→   243→}
   244→   244→
   245→   245→/**
   246→   246→ * Resolve the final assurance items to render for a given product.
   247→   247→ *
   248→   248→ * The single entry point used by `ProductAssuranceGrid`. Components never
   249→   249→ * read `product.assuranceItems` directly — they go through this helper so
   250→   250→ * the merge + dedup behaviour cannot be bypassed.
   251→   251→ */
   252→   252→export function getProductAssuranceItems(product: ProductData): ProductAssurance[] {
   253→   253→  return mergeAssuranceItems(product.assuranceItems);
   254→   254→}
