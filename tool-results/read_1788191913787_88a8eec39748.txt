   267→   tabs remain reachable on narrow screens without

   269→   is auto-scrolled into view via the onValueChange

  1132→/* ────────────────────────────────────────────
  1133→   Main ProductsClient component
  1134→   ────────────────────────────────────────────
  1135→   Single source of truth: `activeCategory` (a
  1136→   ProductTabId). Derived from the URL `?category=`
  1137→   search param on mount and kept in sync with
  1138→   browser back/forward. Tab clicks update both the
  1139→   state and the URL (via router.replace, so tab
  1140→   switches don't pollute history — refresh still
  1141→   restores the selected tab from the URL).
  1142→
  1143→   useSearchParams() requires a <Suspense> boundary
  1144→   in Next.js 13+ when used in a client component
  1145→   rendered by a server component. The default
  1146→   export below wraps the inner component in
  1147→   <Suspense> to satisfy that requirement. */
  1148→
  1149→function ProductsClientInner() {
  1150→  const searchParams = useSearchParams();
  1151→  const router = useRouter();
  1152→  const pathname = usePathname();
  1153→  const tabsListRef = useRef<HTMLDivElement>(null);
  1154→
  1155→  // activeCategory is DERIVED from the URL search param
  1156→  // (single source of truth = the URL). This avoids a
  1157→  // duplicated state copy and the cascading-render
  1158→  // problem of syncing state inside an effect. The URL
  1159→  // is updated by handleTabChange on user clicks, and
  1160→  // browser back/forward naturally updates the URL
  1161→  // (which re-derives activeCategory on the next render).
  1162→  const activeCategory: ProductTabId = resolveTabId(searchParams.get(URL_PARAM));
  1163→
  1164→  // Auto-scroll the active tab into view on mobile so
  1165→  // selecting a tab from another interaction (e.g.
  1166→  // back/forward) keeps the active pill visible in the
  1167→  // horizontally-scrollable strip. Runs after the
  1168→  // active tab changes.
  1169→  useEffect(() => {
  1170→    const list = tabsListRef.current;
  1171→    if (!list) return;
  1172→    const activeBtn = list.querySelector<HTMLButtonElement>(
  1173→      `[data-state="active"]`,
  1174→    );
  1175→    if (activeBtn) {
  1176→      activeBtn.scrollIntoView({
  1177→        behavior: 'smooth',
  1178→        block: 'nearest',
  1179→        inline: 'nearest',
  1180→      });
  1181→    }
  1182→  }, [activeCategory]);
  1183→
  1184→  const handleTabChange = useCallback(
  1185→    (next: ProductTabId) => {
  1186→      // Update the URL with router.replace + scroll:false.
  1187→      // The address bar reflects the selection, refresh
  1188→      // restores it, but no scroll jump or history entry
  1189→      // is created. activeCategory re-derives from the
  1190→      // new searchParams on the next render.
  1191→      const sp = new URLSearchParams(searchParams.toString());
  1192→      sp.set(URL_PARAM, next);
  1193→      router.replace(`${pathname}?${sp.toString()}`, { scroll: false });
  1194→    },
  1195→    [pathname, router, searchParams],
  1196→  );
  1197→
  1198→  return (
  1199→    <div className="min-h-screen flex flex-col bg-be-warm-white">
  1200→      <Header />
  1201→      <main className="flex-1">
  1202→        {/* Breadcrumb */}
  1203→        <div className="container-site page-horizontal-padding pt-5">
  1204→          <Breadcrumb
  1205→            items={[
  1206→              { label: 'Home', href: '/' },
  1207→              { label: 'Products' },
  1208→            ]}
  1209→          />
  1210→        </div>
  1211→
  1212→        {/* 1. Hero (split layout) */}
  1213→        <ProductsHero />
  1214→
  1215→        {/* 2. Category tabs + active category content.
  1216→            The tab strip and the visible content share
  1217→            one source of truth (activeCategory), so the
  1218→            selected tab and displayed family can never
  1219→            disagree. Selecting a tab switches content
  1220→            in place — no anchor scroll, no page jump. */}
  1221→        <ProductCategoryTabs activeId={activeCategory} onChange={handleTabChange} listRef={tabsListRef} />
  1222→
  1223→        {/* Active category content. Each TabsContent only
  1224→            mounts its panel when active (Radix default),
  1225→            keeping the DOM lean and avoiding hidden
  1226→            duplicate IDs. */}
  1227→        <Tabs value={activeCategory} onValueChange={(v) => handleTabChange(v as ProductTabId)} className="flex-col gap-0">
  1228→          <TabsContent value="electrical" className="focus-visible:outline-none">
  1229→            <ElectricalInsulatingMatsSection />
  1230→          </TabsContent>
  1231→          <TabsContent value="waterproofing" className="focus-visible:outline-none">
  1232→            <WaterproofingSection />
  1233→          </TabsContent>
  1234→          <TabsContent value="pvc" className="focus-visible:outline-none">
  1235→            <PvcFlooringSection />
  1236→          </TabsContent>
  1237→          <TabsContent value="other" className="focus-visible:outline-none">
  1238→            <OtherProductsSection />
  1239→          </TabsContent>
  1240→          <TabsContent value="compare" className="focus-visible:outline-none">
  1241→            <ComparisonSection />
  1242→          </TabsContent>
  1243→        </Tabs>
  1244→
  1245→        {/* 7. Selection Guide — always visible below the
  1246→            tab panels (it cross-references all families). */}
  1247→        <SelectionGuideSection />
  1248→        {/* 8. Technical Guidance CTA */}
  1249→        <TechnicalGuidanceCTA />
  1250→      </main>
  1251→      <Footer />
  1252→      <BackToTop />
  1253→      <MobileStickyCTA />
  1254→    </div>
  1255→  );
  1256→}
